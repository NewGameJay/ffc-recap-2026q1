A3. DIRECTLY MH-1 — Email click→subscribe ($20,156 ARR)

Strict filter to MH-published emails only. 23 of 311 unique clickers on Mike's 22 emails became paying subscribers in the window, contributing $1,679.70 MRR ($20,156.40 annualized).

Source query:

WITH mh_campaigns AS (
  SELECT mec.campaign_id
  FROM `...marketing_email` me
  JOIN `...marketing_email_campaign` mec ON mec.marketing_email_id = me.id
  WHERE LOWER(me.published_by_email) LIKE '%@marketerhire.com'
    AND DATE(me.published_at) BETWEEN '2026-02-01' AND '2026-04-29'
),
clickers AS (
  SELECT DISTINCT LOWER(ev.recipient) AS email
  FROM `...email_event` ev
  JOIN mh_campaigns mh ON ev.email_campaign_id = mh.campaign_id
  WHERE ev.type = 'CLICK' AND DATE(ev.created) BETWEEN '2026-02-01' AND '2026-04-29'
)
SELECT COUNT(DISTINCT s.id), ROUND(SUM(s.monthly_recurring_revenue)/100.0 * 12, 2)
FROM clickers cl
JOIN `...customer` cust ON LOWER(cust.email) = cl.email
JOIN `...subscription` s ON s.customer_id = cust.id
WHERE DATE(TIMESTAMP_SECONDS(s.started_at)) BETWEEN '2026-02-01' AND '2026-04-29'
  AND s.deleted = FALSE;
-- Returns: 23 subs / $20,156.40 ARR
